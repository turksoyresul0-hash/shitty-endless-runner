<script type="module">

import * as THREE from
"https://cdn.jsdelivr.net/npm/three@0.180.0/build/three.module.js";

import { PointerLockControls } from
"https://cdn.jsdelivr.net/npm/three@0.180.0/examples/jsm/controls/PointerLockControls.js";


// =========================
// SAHNE
// =========================

const scene = new THREE.Scene();

scene.background = new THREE.Color(0x87ceeb);

scene.fog = new THREE.Fog(0x87ceeb, 20, 150);


// =========================
// KAMERA
// =========================

const camera = new THREE.PerspectiveCamera(
    75,
    window.innerWidth / window.innerHeight,
    0.1,
    1000
);

camera.position.set(0, 2, 5);


// =========================
// RENDER
// =========================

const renderer = new THREE.WebGLRenderer({
    antialias: true
});

renderer.setSize(
    window.innerWidth,
    window.innerHeight
);

renderer.setPixelRatio(
    Math.min(window.devicePixelRatio, 2)
);

document.body.appendChild(renderer.domElement);


// =========================
// IŞIK
// =========================

const ambientLight = new THREE.AmbientLight(
    0xffffff,
    0.7
);

scene.add(ambientLight);


const sun = new THREE.DirectionalLight(
    0xffffff,
    1.5
);

sun.position.set(
    30,
    50,
    20
);

scene.add(sun);


// =========================
// ZEMİN
// =========================

const groundGeometry =
    new THREE.PlaneGeometry(300, 300);

const groundMaterial =
    new THREE.MeshStandardMaterial({
        color: 0x3c7a3c
    });

const ground =
    new THREE.Mesh(
        groundGeometry,
        groundMaterial
    );

ground.rotation.x = -Math.PI / 2;

scene.add(ground);


// =========================
// HARİTA DUVARLARI
// =========================

function createBox(
    x,
    y,
    z,
    width,
    height,
    depth,
    color
) {

    const geometry =
        new THREE.BoxGeometry(
            width,
            height,
            depth
        );

    const material =
        new THREE.MeshStandardMaterial({
            color: color
        });

    const mesh =
        new THREE.Mesh(
            geometry,
            material
        );

    mesh.position.set(
        x,
        y,
        z
    );

    scene.add(mesh);

    return mesh;
}


// Duvarlar

createBox(
    0,
    5,
    -70,
    140,
    10,
    2,
    0x555555
);

createBox(
    0,

    70,
    140,
    10,

    0x555555
);

createBox(

);

createBox(

    0x555555
);


// =========================
// KONTROLLER
// =========================

const controls =
    new PointerLockControls(
        camera,
        document.body
    );

scene.add(controls.object);


// Oyuna tıklayınca mouse kilitle

document.body.addEventListener(
    "click",
    () => {

        controls.lock();

    }
);


// =========================
// HAREKET
// =========================

const keys = {};

document.addEventListener(
    "keydown",
    event => {

        keys[event.code] = true;

        if (event.code === "Space") {
            jump();
        }

        if (event.code === "KeyR") {
            reload();
        }

    }
);


document.addEventListener(
    "keyup",
    event => {

        keys[event.code] = false;

    }
);


// =========================
// FİZİK
// =========================

let velocityY = 0;

let canJump = true;

const gravity = 25;


// =========================
// ZIPLAMA
// =========================

function jump() {

    if (!canJump) return;

    velocityY = 9;

    canJump = false;

}


// =========================
// SİLAH
// =========================

let ammo = 30;

let reserveAmmo = 120;

let shooting = false;


function shoot() {

    if (!controls.isLocked) return;

    if (ammo <= 0) {

        reload();

        return;

    }

    ammo--;

    document.getElementById(
        "ammo"
    ).textContent = ammo;

    // Raycast

    const raycaster =
        new THREE.Raycaster();

    raycaster.setFromCamera(
        new THREE.Vector2(0, 0),
        camera
    );

    const hits =
        raycaster.intersectObjects(
            scene.children,
            true
        );

    if (hits.length > 0) {

        const target =
            hits.find(
                h => h.object.userData.enemy
            );

        if (target) {

            damageEnemy(
                target.object
            );

        }

    }

}


function reload() {

    if (ammo >= 30) return;

    if (reserveAmmo <= 0) return;

    const needed =
        30 - ammo;

    const amount =
        Math.min(
            needed,
            reserveAmmo
        );

    ammo += amount;

    reserveAmmo -= amount;

    document.getElementById(
        "ammo"
    ).textContent = ammo;

}


// Sol tık

document.addEventListener(
    "mousedown",
    event => {

        if (event.button === 0) {

            shoot();

        }

    }
);


// =========================
// DÜŞMAN
// =========================

const enemies = [];


function createEnemy(
    x,
    z
) {

    const geometry =
        new THREE.BoxGeometry(
            1.5,
            2.5,
            1.5
        );

    const material =
        new THREE.MeshStandardMaterial({
            color: 0xaa2222
        });

    const enemy =
        new THREE.Mesh(
            geometry,
            material
        );

    enemy.position.set(
        x,
        1.25,
        z
    );

    enemy.userData.enemy = true;

    enemy.userData.hp = 100;

    scene.add(enemy);

    enemies.push(enemy);

}


createEnemy(
    10,
    -20
);

createEnemy(
    -15,
    -30
);

createEnemy(
    20,
    -40
);


// =========================
// DÜŞMAN HASARI
// =========================

function damageEnemy(enemy) {

    enemy.userData.hp -= 34;

    enemy.material.color.set(
        0xffffff
    );

    setTimeout(() => {

        if (enemy.parent) {

            enemy.material.color.set(
                0xaa2222
            );

        }

    }, 80);


    if (enemy.userData.hp <= 0) {

        scene.remove(enemy);

        const index =
            enemies.indexOf(enemy);

        if (index !== -1) {

            enemies.splice(
                index,
                1
            );

        }

    }

}


// =========================
// DÜŞMAN AI
// =========================

function updateEnemies(delta) {

    const player =
        controls.object.position;

    enemies.forEach(enemy => {

        const direction =
            new THREE.Vector3();

        direction.subVectors(
            player,
            enemy.position
        );

        direction.y = 0;

        const distance =
            direction.length();

        if (distance > 3) {

            direction.normalize();

            enemy.position.add(
                direction.multiplyScalar(
                    delta * 2
                )
            );

        }

    });

}


// =========================
// ANİMASYON
// =========================

const clock =
    new THREE.Clock();


function animate() {

    requestAnimationFrame(
        animate
    );

    const delta =
        Math.min(
            clock.getDelta(),
            0.05
        );


    // -------------------------
    // Hareket
    // -------------------------

    const speed = 8;


    if (controls.isLocked) {

        if (keys["KeyW"]) {

            controls.moveForward(
                speed * delta
            );

        }

        if (keys["KeyS"]) {

            controls.moveForward(
                -speed * delta
            );

        }

        if (keys["KeyA"]) {

            controls.moveRight(
                -speed * delta
            );

        }

        if (keys["KeyD"]) {

            controls.moveRight(
                speed * delta
            );

        }

    }


    // -------------------------
    // Yerçekimi
    // -------------------------

    velocityY -=
        gravity * delta;

    controls.object.position.y +=
        velocityY * delta;


    if (
        controls.object.position.y
        <= 2
    ) {

        controls.object.position.y =
            2;

        velocityY = 0;

        canJump = true;

    }


    // -------------------------
    // Harita sınırı
    // -------------------------

    controls.object.position.x =
        THREE.MathUtils.clamp(
            controls.object.position.x,
            -65,
            65
        );

    controls.object.position.z =
        THREE.MathUtils.clamp(
            controls.object.position.z,
            -65,
            65
        );


    // -------------------------
    // Düşmanlar
    // -------------------------

    updateEnemies(delta);


    renderer.render(
        scene,
        camera
    );

}


animate();


// =========================
// PENCERE BOYUTU
// =========================

window.addEventListener(
    "resize",
    () => {

        camera.aspect =
            window.innerWidth /
            window.innerHeight;

        camera.updateProjectionMatrix();

        renderer.setSize(
            window.innerWidth,
            window.innerHeight
        );

    }
);
